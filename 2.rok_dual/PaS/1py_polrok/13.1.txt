for.sh:

	#!/usr/bin/env bash

	file_content="$(cat file)"

	echo "$file_content" | wc -l

	for line in $file_content
	do
	        echo $line
	done

while.sh
	#!/usr/bin/env bash

	while read line
	do
	        echo "$line"
	done < file
